cmake_minimum_required(VERSION 3.22)
project(home_tunnel_remote_jni LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED HOME_TUNNEL_REMOTE_ROOT)
  message(FATAL_ERROR "Use a verified Home Tunnel shared-core artifact; no separate WebRTC distribution is accepted")
endif()
add_library(home_tunnel_remote SHARED IMPORTED)
set_target_properties(home_tunnel_remote PROPERTIES
  IMPORTED_LOCATION "${HOME_TUNNEL_REMOTE_ROOT}/${ANDROID_ABI}/libhome_tunnel_remote.so")
add_library(home_tunnel_remote_jni SHARED remote_jni.cpp)
target_include_directories(home_tunnel_remote_jni PRIVATE "${HOME_TUNNEL_REMOTE_ROOT}/include")
target_compile_options(home_tunnel_remote_jni PRIVATE -Wall -Wextra -Werror)
target_link_libraries(home_tunnel_remote_jni PRIVATE home_tunnel_remote android)
